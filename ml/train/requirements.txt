# CPU-only torch. The default PyPI wheel pulls the full CUDA/NVIDIA stack (~5-7GB)
# which is useless on our CPU-only VMs. The +cpu local tag exists ONLY on the
# PyTorch CPU index (see --extra-index-url in Dockerfile.train), so pip cannot fall
# back to the CUDA build.
# Pinned WITHOUT the `+cpu` local tag (2026-09-24): the CPU index names the x86_64 wheel
# `2.5.1+cpu` but the aarch64 one plain `2.5.1`, so the tagged pin can never resolve on an
# Apple-silicon Docker build. `==2.5.1` matches both (PEP 440: a local tag satisfies an
# untagged `==`), and Dockerfile.train installs torch from the CPU index FIRST, so x86_64
# (the VM's serve image) still gets exactly `2.5.1+cpu`. Same version everywhere.
torch==2.5.1
numpy>=1.26.0
pandas>=2.1.0
psycopg2-binary>=2.9.9
# SQLAlchemy held below 2.1 (2026-09-25): 2.1.0 makes `postgresql://` resolve to the psycopg 3
# driver, which is not installed here — X8′ seed 3 failed on "No module named 'psycopg'"
# the first time an image was built after the 2.1.0 release. 2.0.x keeps psycopg2.
SQLAlchemy>=2.0.0,<2.1
scikit-learn>=1.3.0
lightgbm>=4.1.0  # GBT diagnostic baseline (gbt_baseline.py); not a serve dependency
requests>=2.31.0
pyarrow>=14.0.0  # parquet engine for eval_m2.py --dump-preds (falls back to csv.gz if absent)
